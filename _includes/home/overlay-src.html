{%- comment -%} Resolves a page's door_image (a door's plate, for a page with no hero of its own), else its header.overlay_image, to a URL with the same rules as page__hero.html: absolute as-is · root-relative via base · else under /images/. Usage: {% include home/overlay-src.html page=dp %} {%- endcomment -%} {%- assign ov = include.page.door_image | default: include.page.header.overlay_image -%} {%- if ov contains "://" -%}{{ ov }}{%- else -%}{%- assign fc = ov | slice: 0, 1 -%}{%- if fc == "/" -%}{{ ov | relative_url }}{%- else -%}{{ ov | prepend: "/images/" | relative_url }}{%- endif -%}{%- endif -%}
