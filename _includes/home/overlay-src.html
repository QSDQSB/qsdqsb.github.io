{%- comment -%} Resolves a destination page's header.overlay_image to a usable URL, mirroring the same rules as _includes/page__hero.html so the N° II plate reuses each page's own hero image: "http(s)://…" as-is · "/…" prepended with base only · otherwise prepended with "/images/". Usage: {% include home/overlay-src.html page=door_posts %} {%- endcomment -%} {%- assign ov = include.page.header.overlay_image -%} {%- if ov contains "://" -%}{{ ov }}{%- else -%}{%- assign fc = ov | slice: 0, 1 -%}{%- if fc == "/" -%}{{ ov | relative_url }}{%- else -%}{{ ov | prepend: "/images/" | relative_url }}{%- endif -%}{%- endif -%}
